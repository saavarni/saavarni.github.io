# cleaning
ny_noaa <- 
  ny_noaa |> 
  drop_na() |>
  mutate(
    year = year(date),
    month = month(date),
    day = day(date),
    tmax = as.numeric(tmax) / 10,
    tmin = as.numeric(tmin) / 10, 
    prcp = prcp / 10
  )

# sampling 
set.seed(7)
noaa_sample = 
  ny_noaa |> 
  filter(year >= 2000) |>     
  slice_sample(n = 1000)

Column

Chart A

noaa_sample |> 
  mutate(text_label = str_c("Precip: ", prcp, " mm\nMonth: ", month)) |> 
  plot_ly(
    x = ~tmin, y = ~tmax, color = ~month, text = ~text_label,
    alpha = .5, type = "scatter", mode = "markers")

Column

Chart B

noaa_sample |> 
  mutate(month = fct_reorder(month.abb[month], tmax)) |> 
  plot_ly(y = ~tmax, color = ~month, type = "box", colors = "viridis")

Chart C

noaa_sample |> 
  filter(snow > 0) |> 
  mutate(month = month.abb[month]) |> 
  count(month) |> 
  mutate(month = fct_reorder(month, n)) |> 
  plot_ly(x = ~month, y = ~n, color = ~month, type = "bar", colors = "viridis")
